function loadBillData() { try { const p = new URLSearchParams(window.location.search); const bld = (p.get('building') || '38HD').trim(); const mon = (p.get('month') || '-').trim(); const rom = (p.get('room') || '-').trim(); const ten = decodeURIComponent(p.get('tenant') || '-').trim(); groupLink = decodeURIComponent(p.get('link') || '').trim(); const parseSafeInt = (val, defaultVal = 0) => { const parsed = parseInt(val); return isNaN(parsed) ? defaultVal : parsed; }; const eO = parseSafeInt(p.get('e_old')); const eN = parseSafeInt(p.get('e_new')); const eU = Math.max(0, eN - eO); const ePrice = parseSafeInt(p.get('e_price'), 4000); let eAmt = parseSafeInt(p.get('e_amount'), eU * ePrice); const rnt = parseSafeInt(p.get('rent')); const wat = parseSafeInt(p.get('water')); const wif = parseSafeInt(p.get('wifi')); const srv = parseSafeInt(p.get('s_fee')); // MỤC KHÁC 1 & 2 const oth = parseSafeInt(p.get('other')); const otherNote = decodeURIComponent(p.get('other_note') || '').replace(/^undefined$|^null$/, '').trim(); const oth2 = parseSafeInt(p.get('other2')); const other2Desc = decodeURIComponent(p.get('other2_desc') || '').replace(/^undefined$|^null$/, '').trim(); const other2Note = decodeURIComponent(p.get('other2_note') || '').replace(/^undefined$|^null$/, '').trim(); // KỲ HẠN const rentPeriod = decodeURIComponent(p.get('rent_period') || '').replace(/^undefined$|^null$/, '').trim(); const elecPeriod = decodeURIComponent(p.get('elec_period') || '').replace(/^undefined$|^null$/, '').trim(); const total = rnt + eAmt + wat + wif + srv + oth + oth2; const year = new Date().getFullYear(); const code = `${bld}P${rom}T${mon}Y${year}`; globalRoom = rom; globalMonth = mon; globalCode = code; globalTotal = fmt(total); const c = cfg[bld] || cfg["38HD"]; const lang = c.isEn ? t.en : t.vi; document.getElementById('v_brand').innerText = c.name; document.getElementById('v_addr').innerText = c.addr; document.getElementById('v_bank').innerText = c.bank + " Bank"; document.getElementById('v_acc').innerText = c.acc; document.getElementById('v_owner').innerText = c.owner; document.getElementById('v_code').innerText = code; for(let k in lang) { const el = document.getElementById('lbl_' + k); if(el) el.innerText = lang[k]; } document.getElementById('v_title').innerText = lang.main; document.getElementById('v_subtitle').innerText = lang.sub; document.getElementById('v_month').innerText = `${mon} / ${year}`; document.getElementById('v_room').innerText = rom; document.getElementById('v_tenant').innerText = ten; document.getElementById('v_rent_amt').innerText = fmt(rnt); document.getElementById('v_elec_amt').innerText = fmt(eAmt); document.getElementById('v_water_amt').innerText = fmt(wat); document.getElementById('v_wifi_amt').innerText = fmt(wif); document.getElementById('v_serv_amt').innerText = fmt(srv); document.getElementById('v_total').innerText = globalTotal; // XỬ LÝ KỲ HẠN let showPeriodBox = false; if (rentPeriod) { document.getElementById('v_rent_period').innerText = rentPeriod; document.getElementById('row_rent_p').style.display = 'flex'; showPeriodBox = true; } if (elecPeriod) { document.getElementById('v_elec_period').innerText = elecPeriod; document.getElementById('row_elec_p').style.display = 'flex'; showPeriodBox = true; } if (showPeriodBox) { document.getElementById('box_period').style.display = 'block'; } // XỬ LÝ MỤC KHÁC 1 document.getElementById('v_other_amt').innerText = fmt(oth); const elOtherNote = document.getElementById('v_other_note'); if (otherNote) { elOtherNote.innerText = `(${otherNote})`; elOtherNote.style.display = 'block'; } // XỬ LÝ MỤC KHÁC 2 if (oth2 > 0 || other2Note) { const rowOther2 = document.getElementById('row_other2'); rowOther2.style.display = 'table-row'; if (other2Desc) { document.getElementById('lbl_other2').innerText = other2Desc; } document.getElementById('v_other2_amt').innerText = fmt(oth2); if (other2Note) { const elOther2Note = document.getElementById('v_other2_note'); elOther2Note.innerText = `(${other2Note})`; elOther2Note.style.display = 'block'; } } document.getElementById('v_e_old').innerText = fmt(eO); document.getElementById('v_e_new').innerText = fmt(eN); document.getElementById('v_e_used').innerText = fmt(eU); document.getElementById('v_e_subtotal').innerText = fmt(eAmt); // TẠO QR VIETQR const bankClean = encodeURIComponent(c.bank.trim()); const accClean = encodeURIComponent(c.acc.trim()); const ownerClean = encodeURIComponent(c.owner.trim()); const qrUrl = `https://img.vietqr.io/image/${bankClean}-${accClean}-qr_only.png?amount=${total}&addInfo=${encodeURIComponent(code)}&accountName=${ownerClean}`; const qrImg = document.getElementById('v_qr'); qrImg.src = qrUrl; qrImg.onerror = function() { qrImg.src = `https://api.qrserver.com/v1/create-qr-code/?size=150x150&data=${encodeURIComponent(`2|99|${c.acc}|${c.bank}|${total}|${code}`)}`; }; } catch (err) { console.error("Lỗi script:", err); showToast("Lỗi tải dữ liệu!", true); } }
